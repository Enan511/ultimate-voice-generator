cmake_minimum_required(VERSION 3.22)
project(uvg_native VERSION 3.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH "armv8.2-a+dotprod" CACHE STRING "" FORCE)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(ENGINE STREQUAL "qwen")
  set(QWEN ${ROOT}/vendor/qwentts)
  add_compile_definitions(GGML_MAX_NAME=128 QWEN_STATIC)
  add_subdirectory(${QWEN}/ggml ggml)
  file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/version.h "#define QWEN_VERSION \"6fae929 Android ARM64\"\n")
  add_library(uvg_qwen SHARED qwen-jni.cpp ${QWEN}/src/qwen.cpp ${QWEN}/src/pipeline-tts.cpp ${QWEN}/src/pipeline-codec.cpp)
  target_include_directories(uvg_qwen PRIVATE ${QWEN}/src ${QWEN} ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(uvg_qwen PRIVATE ggml ggml-base ggml-cpu log)
  target_link_options(uvg_qwen PRIVATE -Wl,-Bsymbolic -Wl,--exclude-libs,ALL)
elseif(ENGINE STREQUAL "whisper")
  set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  add_subdirectory(${ROOT}/vendor/whisper whisper)
  add_library(uvg_whisper SHARED whisper-jni.cpp)
  target_link_libraries(uvg_whisper PRIVATE whisper log)
  target_link_options(uvg_whisper PRIVATE -Wl,-Bsymbolic -Wl,--exclude-libs,ALL)
else()
  set(BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
  set(BUILD_DOCS OFF CACHE BOOL "" FORCE)
  set(INSTALL_MANPAGES OFF CACHE BOOL "" FORCE)
  set(BUILD_CXXLIBS OFF CACHE BOOL "" FORCE)
  set(WITH_OGG OFF CACHE BOOL "" FORCE)
  add_subdirectory(${ROOT}/vendor/flac flac)
  # LGPL libraries remain separate, replaceable shared objects.
  set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
  set(SOUNDTOUCH_DLL OFF CACHE BOOL "" FORCE)
  set(SOUNDSTRETCH OFF CACHE BOOL "" FORCE)
  add_subdirectory(${ROOT}/vendor/soundtouch soundtouch)
  file(GLOB LAME_SOURCES ${ROOT}/vendor/lame/libmp3lame/*.c)
  list(REMOVE_ITEM LAME_SOURCES ${ROOT}/vendor/lame/libmp3lame/mpglib_interface.c)
  add_library(mp3lame SHARED ${LAME_SOURCES})
  target_compile_definitions(mp3lame PRIVATE HAVE_CONFIG_H)
  target_include_directories(mp3lame PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/lame-config ${ROOT}/vendor/lame/include ${ROOT}/vendor/lame/libmp3lame)
  set_target_properties(mp3lame SoundTouch PROPERTIES C_VISIBILITY_PRESET default CXX_VISIBILITY_PRESET default)
  set_target_properties(SoundTouch PROPERTIES VERSION "" SOVERSION "")
  add_library(uvg_audio SHARED audio-jni.cpp)
  target_include_directories(uvg_audio PRIVATE ${ROOT}/vendor/lame/include)
  target_link_libraries(uvg_audio PRIVATE FLAC SoundTouch mp3lame log)
endif()
